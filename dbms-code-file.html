PR 5 Constraints — CollegeDB
🏢Create Database & Dept TablePrimary Key, Not Null, Unique, Default
CREATE DATABASE CollegeDB;
USE CollegeDB;

CREATE TABLE Dept (
  DeptID   INT PRIMARY KEY,
  DeptName VARCHAR(50) NOT NULL UNIQUE,
  Location VARCHAR(50) DEFAULT 'Main Building'
);
🎓Students Table — All Constraints + FK CascadeCheck, Unique, FK, Cascade
CREATE TABLE Students (
  StudentID INT PRIMARY KEY,
  Name      VARCHAR(50) NOT NULL,
  Age       INT CHECK (Age >= 18),
  email     VARCHAR(100) UNIQUE,
  DeptID    INT,
  CONSTRAINT FK_Dept
    FOREIGN KEY (DeptID)
    REFERENCES Dept(DeptID)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

FUNCTION Function — calc_bonus
🧮calc_bonus Function + UsageFUNCTION • DETERMINISTIC
DELIMITER //
CREATE FUNCTION calc_bonus(sal INT)
RETURNS INT
DETERMINISTIC
BEGIN
  DECLARE bonus INT;
  SET bonus = sal * 0.10;
  RETURN bonus;
END //
DELIMITER ;

-- Test with a value:
SELECT calc_bonus(50000);

-- Use with employee table:
SELECT name, salary, calc_bonus(salary) AS bonus
FROM employee;
TRIGGERS Triggers — trigger_demo
🏗️1) Create Tables (main, backup, update_log, employees)Setup
CREATE DATABASE trigger_demo;
USE trigger_demo;

CREATE TABLE main (
  id     INT PRIMARY KEY,
  salary INT
);
CREATE TABLE backup (
  id     INT,
  salary INT
);
CREATE TABLE update_log (
  id         INT,
  old_salary INT,
  new_salary INT
);
CREATE TABLE employees (
  id     INT PRIMARY KEY,
  salary INT
);
⬇️2) BEFORE INSERT Trigger + TestSalary cannot be negative
DELIMITER //
CREATE TRIGGER t_before_insert
BEFORE INSERT ON main
FOR EACH ROW
BEGIN
  IF NEW.salary < 0 THEN
    SET NEW.salary = 0;
  END IF;
END //
DELIMITER ;

-- Test (negative becomes 0):
INSERT INTO main VALUES (1, -5000);
SELECT * FROM main;
⬆️3) AFTER INSERT Trigger + TestCopy to backup
DELIMITER //
CREATE TRIGGER t_after_insert
AFTER INSERT ON main
FOR EACH ROW
BEGIN
  INSERT INTO backup(id, salary)
  VALUES (NEW.id, NEW.salary);
END //
DELIMITER ;

-- Test:
INSERT INTO main VALUES (2, 5000), (3, 10000);
SELECT * FROM backup;
⚠️4) BEFORE UPDATE Trigger + TestSalary min = 5000
DELIMITER //
CREATE TRIGGER before_update_salary
BEFORE UPDATE ON employees
FOR EACH ROW
BEGIN
  IF NEW.salary < 5000 THEN
    SIGNAL SQLSTATE '45000'
    SET MESSAGE_TEXT = 'Salary cannot be less than 5000';
  END IF;
END //
DELIMITER ;

-- Test:
INSERT INTO employees VALUES (1, 7000);
UPDATE employees SET salary = 8000 WHERE id = 1;  -- OK
UPDATE employees SET salary = 3000 WHERE id = 1;  -- ERROR
📋5) AFTER UPDATE Trigger + TestLog old & new salary
DELIMITER //
CREATE TRIGGER after_update_main
AFTER UPDATE ON main
FOR EACH ROW
BEGIN
  INSERT INTO backup(id, salary)
  VALUES (OLD.id, OLD.salary);
  INSERT INTO update_log(id, old_salary, new_salary)
  VALUES (OLD.id, OLD.salary, NEW.salary);
END //
DELIMITER ;

-- Test:
UPDATE main SET salary = 20000 WHERE id = 2;
SELECT * FROM backup;
SELECT * FROM update_log;
